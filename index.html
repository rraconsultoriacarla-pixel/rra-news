function renderNews(filter = 'all') { const grid = document.getElementById('newsGrid'); grid.innerHTML = ''; const normalizedFilter = filter.trim().toLowerCase(); const filteredNews = normalizedFilter === 'all' ? newsData : newsData.filter(item => item.category && item.category.trim().toLowerCase() === normalizedFilter); if (filteredNews.length === 0) { grid.innerHTML = '

Nenhuma notícia encontrada para esta categoria.

'; return; } filteredNews.forEach(item => { const card = document.createElement('div'); card.className = 'news-card'; card.innerHTML = `
${item.category || 'Geral'}

${item.title}

${item.summary}

`; grid.appendChild(card); }); } function filterNews(categoryName, buttonElement) { document.querySelectorAll('.filter-btn').forEach(btn => { btn.classList.remove('active'); }); // Se o botão foi passado no clique, ativa ele. Caso contrário busca pelo texto if(buttonElement) { buttonElement.classList.add('active'); } else { document.querySelectorAll('.filter-btn').forEach(btn => { if(btn.textContent.trim().toLowerCase() === categoryName.toLowerCase()) { btn.classList.add('active'); } }); } renderNews(categoryName); }
